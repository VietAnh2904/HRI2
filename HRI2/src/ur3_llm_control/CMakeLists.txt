cmake_minimum_required(VERSION 3.16)
project(ur3_llm_control)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
add_library(skill_grasp SHARED src/skill_grasp.cpp)
target_compile_features(skill_grasp PUBLIC cxx_std_17)
target_include_directories(skill_grasp PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(skill_grasp ${GAZEBO_LIBRARIES})
ament_target_dependencies(skill_grasp gazebo_ros rclcpp std_srvs)
install(TARGETS skill_grasp LIBRARY DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/llm_planner scripts/skill_executor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config urdf worlds prompts DESTINATION share/${PROJECT_NAME})
ament_package()
